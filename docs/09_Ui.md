# 🎨 UI_DESIGN_SYSTEM.md — Design System & Component Library

## Design Philosophy

**"Functional Elegance"** — Every pixel serves a purpose. Clean, data-dense interfaces that are beautiful under pressure. Inspired by Linear, Vercel, and Stripe Dashboards.

### DataTable (Most Used)
```tsx
// packages/ui/components/composed/DataTable/DataTable.tsx

interface DataTableProps<TData, TValue> {
  columns: ColumnDef<TData, TValue>[];
  data: TData[];
  isLoading?: boolean;
  searchColumn?: string;
  filterOptions?: FilterOption[];
  onRowClick?: (row: TData) => void;
  bulkActions?: BulkAction<TData>[];
  exportable?: boolean;
  pagination?: PaginationState;
  onPaginationChange?: (state: PaginationState) => void;
}

export function DataTable<TData, TValue>({
  columns, data, isLoading, searchColumn,
  filterOptions, onRowClick, bulkActions,
  exportable, pagination, onPaginationChange,
}: DataTableProps<TData, TValue>) {
  const [rowSelection, setRowSelection] = useState({});
  const [columnFilters, setColumnFilters] = useState<ColumnFiltersState>([]);
  const [sorting, setSorting] = useState<SortingState>([]);

  const table = useReactTable({
    data,
    columns,
    state: { rowSelection, columnFilters, sorting, pagination },
    onRowSelectionChange: setRowSelection,
    onColumnFiltersChange: setColumnFilters,
    onSortingChange: setSorting,
    onPaginationChange,
    getCoreRowModel: getCoreRowModel(),
    getFilteredRowModel: getFilteredRowModel(),
    getSortedRowModel: getSortedRowModel(),
    manualPagination: true,
  });

  return (
    <div className="space-y-4">
      <DataTableToolbar table={table} searchColumn={searchColumn}
        filterOptions={filterOptions} exportable={exportable}
        bulkActions={bulkActions} selectedRows={Object.keys(rowSelection)} />
      
      <div className="rounded-lg border border-border overflow-hidden">
        <Table>
          <TableHeader>
            {table.getHeaderGroups().map(headerGroup => (
              <TableRow key={headerGroup.id} className="bg-muted/50">
                {headerGroup.headers.map(header => (
                  <TableHead key={header.id}>
                    <DataTableColumnHeader header={header} />
                  </TableHead>
                ))}
              </TableRow>
            ))}
          </TableHeader>
          <TableBody>
            {isLoading ? (
              Array.from({ length: 10 }).map((_, i) => (
                <TableRow key={i}>
                  {columns.map((_, j) => (
                    <TableCell key={j}>
                      <div className="h-4 bg-muted rounded animate-shimmer bg-gradient-to-r from-muted via-muted/50 to-muted bg-[length:200%_100%]" />
                    </TableCell>
                  ))}
                </TableRow>
              ))
            ) : table.getRowModel().rows.map(row => (
              <TableRow key={row.id}
                className={cn(onRowClick && "cursor-pointer hover:bg-muted/50 transition-colors")}
                onClick={() => onRowClick?.(row.original)}>
                {row.getVisibleCells().map(cell => (
                  <TableCell key={cell.id}>
                    {flexRender(cell.column.columnDef.cell, cell.getContext())}
                  </TableCell>
                ))}
              </TableRow>
            ))}
          </TableBody>
        </Table>
      </div>

      <DataTablePagination table={table} />
    </div>
  );
}
```

### StatCard
```tsx
// packages/ui/components/composed/Stats/StatCard.tsx

interface StatCardProps {
  title: string;
  value: string | number;
  change?: { value: number; period: string };
  icon: LucideIcon;
  description?: string;
  isLoading?: boolean;
  trend?: 'up' | 'down' | 'neutral';
  formatter?: (val: number) => string;
}

export function StatCard({ title, value, change, icon: Icon, isLoading }: StatCardProps) {
  if (isLoading) return <StatCardSkeleton />;
  
  const isPositive = change && change.value >= 0;
  
  return (
    <Card className="relative overflow-hidden hover:shadow-md transition-shadow">
      <CardContent className="p-6">
        <div className="flex items-start justify-between">
          <div className="space-y-1">
            <p className="text-sm font-medium text-muted-foreground">{title}</p>
            <p className="text-3xl font-bold tracking-tight">{value}</p>
          </div>
          <div className="p-2 bg-primary/10 rounded-lg">
            <Icon className="h-5 w-5 text-primary" />
          </div>
        </div>
        
        {change && (
          <div className="flex items-center gap-1 mt-4">
            {isPositive
              ? <TrendingUp className="h-4 w-4 text-success" />
              : <TrendingDown className="h-4 w-4 text-destructive" />
            }
            <span className={cn(
              "text-sm font-medium",
              isPositive ? "text-success" : "text-destructive"
            )}>
              {isPositive ? '+' : ''}{change.value.toFixed(1)}%
            </span>
            <span className="text-sm text-muted-foreground">vs {change.period}</span>
          </div>
        )}
      </CardContent>
      
      {/* Subtle gradient accent */}
      <div className="absolute bottom-0 left-0 right-0 h-1 bg-gradient-to-r from-primary/60 to-primary/20" />
    </Card>
  );
}
```

## 7. Storefront Theme System

```typescript
// Tenant-customizable theme applied at build/request time

interface StorefrontTheme {
  colors: {
    primary:    string;  // CSS HSL or hex
    secondary:  string;
    accent:     string;
    background: string;
    text:       string;
    border:     string;
  };
  typography: {
    headingFont: string;   // Google Fonts name
    bodyFont:    string;
    scale:       'compact' | 'default' | 'relaxed';
  };
  layout: {
    containerWidth: number;   // px, default 1280
    borderRadius:   'none' | 'sm' | 'md' | 'lg' | 'full';
    headerStyle:    'minimal' | 'standard' | 'mega';
  };
  components: {
    productCard:  'grid' | 'list' | 'compact';
    buttonStyle:  'filled' | 'outlined' | 'ghost';
  };
}

// Injected as CSS custom properties on <html>
function themeToCSS(theme: StorefrontTheme): string {
  return `
    :root {
      --primary: ${theme.colors.primary};
      --font-heading: "${theme.typography.headingFont}", sans-serif;
      --radius: ${radiusMap[theme.layout.borderRadius]};
    }
  `;
}
```

## 8. Responsive Breakpoints

```
xs:  < 480px   — Mobile small
sm:  ≥ 640px   — Mobile large
md:  ≥ 768px   — Tablet
lg:  ≥ 1024px  — Desktop small
xl:  ≥ 1280px  — Desktop
2xl: ≥ 1536px  — Wide desktop
```

**Mobile-first, POS-optimized:**
- POS interface optimized for 10" tablets (1024x768)
- Admin dashboard — min width 768px (tablet sidebar collapses)
- Storefront — fully responsive from 320px
- Mobile app — React Native Dimensions API
